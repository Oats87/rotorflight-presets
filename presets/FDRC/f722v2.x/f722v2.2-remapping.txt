#$ DESCRIPTION: Preset that contains the remapping options for a FLYDRAGON V2.2 such as usable RPM port and servo 5.

#$ OPTION_GROUP BEGIN: (EXCLUSIVE) RPM Port
    #$ OPTION BEGIN (CHECKED): RPM-E
    resource FREQ 1 A01
    #$ OPTION END
    #$ OPTION BEGIN (UNCHECKED): RPM-S
    resource FREQ 1 A15
    #$ OPTION END
#$ OPTION_GROUP END

#$ OPTION_GROUP BEGIN: (EXCLUSIVE) Servo 5 Port
    #$ OPTION BEGIN (UNCHECKED): F.Port
    resource SERIAL_TX 2 none
    resource SERVO 5 A02
    timer A02 AF3
    mixer input CH10 -1000 1000 1000
    mixer rule 10 set CH10 S5 1000 0
    #$ OPTION END
    #$ OPTION BEGIN (UNCHECKED): S.BUS
    resource SERIAL_RX 2 none
    resource SERVO 5 A03
    timer A03 AF3
    mixer input CH10 -1000 1000 1000
    mixer rule 10 set CH10 S5 1000 0
    #$ OPTION END
#$ OPTION_GROUP END
